// Scoring Rule Implementation for Detecting Incorrect Defender Positions
// This query identifies players who were incorrectly assigned as defenders

MATCH (p:Player)-[:PLAYS_AS]->(pos:Position)
WITH p, COLLECT(DISTINCT pos.name) AS positions
WHERE SIZE(positions) = 2 AND 'DEF' IN positions

// Get all fixtures for these players and calculate defender score
MATCH (p)-[r:PLAYED_IN]->(f:Fixture)
WITH p, f, r,
  // General scoring (applies to all positions)
  CASE 
    WHEN r.minutes > 0 AND r.minutes < 60 THEN 1
    WHEN r.minutes >= 60 THEN 2
    ELSE 0
  END +
  (r.assists * 3) +
  (r.penalties_missed * -2) +
  (r.own_goals * -2) +
  (r.yellow_cards * -1) +
  (r.red_cards * -3) +
  // Defender-specific scoring
  (r.goals_scored * 6) +
  CASE 
    WHEN r.clean_sheets = 1 AND r.minutes >= 60 THEN 4
    ELSE 0
  END +
  (toInteger(r.goals_conceded / 2) * -1) AS def_score

// Collect all fixtures and filter for incorrect ones
WITH p,
     COLLECT({
       def_score: def_score,
       season: f.season,
       fixture_number: f.fixture_number,
       matches_def: def_score = r.total_points,
       total_points: r.total_points
     }) AS all_fixture_details

// Filter to only incorrect fixtures
WITH p,
     SIZE(all_fixture_details) AS total_fixtures,
     [fd IN all_fixture_details WHERE fd.matches_def = false] AS incorrect_fixtures

// Only return players who have incorrect fixtures
WHERE SIZE(incorrect_fixtures) > 0

RETURN p.player_name AS player,
       SIZE(incorrect_fixtures) AS not_def_fixtures,
       total_fixtures,
       incorrect_fixtures AS fixture_details
ORDER BY p.player_name
